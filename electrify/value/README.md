Unit Equalities
---------------


### Reciprocals

    1 = [Hz "frequency"] * [s "second"]
    1 = [Ω "resistance"] * [S "conductance"]

### Ampere (A)
Measure of current.

    [A "current"] = [C "electric charge"] / [s "second"]
    [A "current"] = [C "electric charge"] * [Hz "frequency"]
    [A "current"] = [J "energy"] / [Wb "magnetic flux"]
    [A "current"] = [S "conductance"] * [V "voltage"]
    [A "current"] = [V "voltage"] / [Ω "resistance"]
    [A "current"] = [W "power"] / [V "voltage"]
    [A "current"] = [Wb "magnetic flux"] / [H "inductance"]

### Coulomb (C)
Measure of electric charge.

    [C "electric charge"] = [A "current"] * [s "second"]
    [C "electric charge"] = [F "capacitance"] * [V "voltage"]
    [C "electric charge"] = [J "energy"] / [V "voltage"]
    [C "electric charge"] = [Wb "magnetic flux"] / [Ω "resistance"]

### Farad (F)
Measure of capacitance.

    [F "capacitance"] = [C "electric charge"] / [V "voltage"]
    [F "capacitance"] = [S "conductance"] * [s "second"]
    [F "capacitance"] = [S "conductance"] / [Hz "frequency"]
    [F "capacitance"] = [s "second"] / [Ω "resistance"]
    [F "capacitance"] = [A "current"] * [s "seconds"] / [V "voltage"]
    [F "capacitance"] = [C "charge"]^2 / [J "energy"]
    [F "capacitance"] = [J "energy"] / [V "voltage"]^2
    [F "capacitance"] = [s "seconds"]^2 / [H "inductance"]
    [F "capacitance"] = [W "power"] * [s "seconds"] / [V "voltage"]^2

### Henry (H)
Measure of inductance.

    [H "inductance"] = [Ω "resistance"] * [s "second"]
    [H "inductance"] = [Ω "resistance"] / [Hz "frequency"]
    [H "inductance"] = [Wb "magnetic flux"] / [A "current"]
    [H "inductance"] = [s "second"] / [S "conductance"]
    [H "inductance"] = [Wb "magnetic flux"] / [A "current"]
    [H "inductance"] = [J "energy"] / [A "current"]^2
    [H "inductance"] = [s "seconds"]^2 / [F "capacitance]
    [H "inductance"] = [V "voltage"] * [s "seconds"] / [A "current"]

### Hertz (Hz)
Measure of frequency.

    [Hz "frequency"] = 1 / [s "second"]
    [Hz "frequency"] = [Ω "resistance"] / [H "inductance"]
    [Hz "frequency"] = [S "conductance"] / [F "capacitance"]

### Joule (J)
Measure of energy.

    [J "energy"] = [A "current"] * [Wb "magnetic flux"]
    [J "energy"] = [C "electric charge"] * [V "voltage"]
    [J "energy"] = [W "power"] * [s "second"]

### Ohm (Ω)
Measure of resistance, reactance, or impedance.

    [Ω "resistance"] = 1 / [S "conductance"]
    [Ω "resistance"] = [H "inductance"] * [Hz "frequency"]
    [Ω "resistance"] = [H "inductance"] / [s "second"]
    [Ω "resistance"] = [V "voltage"] / [A "current"]
    [Ω "resistance"] = [Wb "magnetic flux"] / [C "electric charge"]
    [Ω "resistance"] = [s "second"] / [F "capacitance"]
    [Ω "resistance"] = ([J "energy"] * [s "seconds"]) / [C "charge"]^2
    [Ω "resistance"] = [V "volts"]^2 / [W "power"]
    [Ω "resistance"] = [W "power"] / [A "current"]^2
    [Ω "resistance"] = [J "energy"] / ([s "seconds"] * [A "current"]^2)

### Siemens (S)
Measure of conductance, susceptance, or admittance.

    [S "conductance"] = 1 / [Ω "resistance"]
    [S "conductance"] = [A "current"] / [V "voltage"]
    [S "conductance"] = [F "capacitance"] * [Hz "frequency"]
    [S "conductance"] = [F "capacitance"] / [s "second"]
    [S "conductance"] = [s "second"] / [H "inductance"]

### Volt (V)
Measure of electric potential difference / voltage.

    [V "voltage"] = [A "current"] * [Ω "resistance"]
    [V "voltage"] = [A "current"] / [S "conductance"]
    [V "voltage"] = [C "electric charge"] / [F "capacitance"]
    [V "voltage"] = [J "energy"] / [C "electric charge"]
    [V "voltage"] = [W "power"] / [A "current"]
    [V "voltage"] = [Wb "magnetic flux"] / [s "second"]
    [V "voltage"] = [Wb "magnetic flux"] [Hz "frequency"]
    [V "voltage"] = [Wb "magnetic flux"] / [s "seconds"]

### Watt (W)
Measure of power.

    [W "power"] = [A "current"] * [V "voltage"]
    [W "power"] = [J "energy"] / [s "second"]
    [W "power"] = [J "energy"] [Hz "frequency"]
    [W "power"] = [A "current"]^2 * [R "resistance"]
    [W "power"] = [V "voltage"]^2 / [R "resistance"]

### Weber (Wb)
Measure of magnetic flux.

    [Wb "magnetic flux"] = [A "current"] * [H "inductance"]
    [Wb "magnetic flux"] = [C "electric charge"] * [Ω "resistance"]
    [Wb "magnetic flux"] = [J "energy"] / [A "current"]
    [Wb "magnetic flux"] = [V "voltage"] * [s "second"]
    [Wb "magnetic flux"] = [T "magnetic flux density"] / [m "meter"]^2

### Second (s)
Measure of time.

    [s "second"] = 1 / [Hz "frequency"]
    [s "second"] = [C "electric charge"] / [A "current"]
    [s "second"] = [F "capacitance"] * [Ω "resistance"]
    [s "second"] = [F "capacitance"] / [S "conductance"]
    [s "second"] = [H "inductance"] * [S "conductance"]
    [s "second"] = [H "inductance"] / [Ω "resistance"]
    [s "second"] = [J "energy"] / [W "power"]
    [s "second"] = [Wb "magnetic flux"] / [V "voltage"]
